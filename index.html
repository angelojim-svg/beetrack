import { serve } from "https://deno.land/std@0.168.0/http/server.ts" const RESEND_API_KEY = Deno.env.get('RESEND_API_KEY') serve(async (req) => { // Gestion de la vérification CORS si nécessaire if (req.method === 'OPTIONS') { return new Response('ok', { headers: { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type' } }) } try { // Vérification de la présence d'un body const bodyText = await req.text() if (!bodyText) { return new Response(JSON.stringify({ error: "Empty request body" }), { status: 400, headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' } }) } const { user_email, race_title, alert_type, race_date, opening_date } = JSON.parse(bodyText) if (!user_email || !race_title) { return new Response(JSON.stringify({ error: "Missing required fields: user_email or race_title" }), { status: 400, headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' } }) } let subject = "" let htmlContent = "" // CAS 1 : Inscription alerte date (TBD) if (alert_type === 'tbd_confirmation') { subject = `BeeTrack - Confirmation d'alerte pour ${race_title}` htmlContent = `

Bonjour,

Votre demande d'alerte pour ${race_title} a bien été prise en compte.

Vous serez informé de la date de l'événement par mail et pourrez la consulter sur BeeTrack.

À bientôt,
L'équipe BeeTrack

` } else if (alert_type === 'tbd_event_date_known') { subject = `BeeTrack - La date de ${race_title} est connue !` htmlContent = `

Bonjour,

La date de l'épreuve ${race_title} est fixée au ${race_date || 'prochainement'}.

Vous pouvez dès à présent la consulter sur BeeTrack.

À bientôt,
L'équipe BeeTrack

` } // CAS 2 : Inscription alerte ouverture inscription (Opening Soon) else if (alert_type === 'opening_soon_confirmation') { subject = `BeeTrack - Confirmation d'alerte ouverture pour ${race_title}` htmlContent = `

Bonjour,

Votre demande d'alerte pour ${race_title} a bien été prise en compte.

Vous serez informé de la date de l'ouverture des inscriptions par mail et pourrez la consulter sur BeeTrack.

À bientôt,
L'équipe BeeTrack

` } else if (alert_type === 'opening_soon_reminder_eve') { subject = `BeeTrack - Rappel : Ouverture des inscriptions pour ${race_title} demain !` htmlContent = `

Bonjour,

Les inscriptions pour l'épreuve ${race_title} ouvrent demain (${opening_date || ''}) !

Vous pouvez la consulter sur BeeTrack.

À bientôt,
L'équipe BeeTrack

` } else { subject = `BeeTrack - Alerte ${race_title}` htmlContent = `

Bonjour, mise à jour concernant ${race_title}. Consultez BeeTrack.

` } // Appel à l'API Resend const res = await fetch('https://api.resend.com/emails', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${RESEND_API_KEY}` }, body: JSON.stringify({ from: 'BeeTrack ', to: [user_email], subject: subject, html: htmlContent }) }) const resData = await res.json() return new Response(JSON.stringify({ success: true, data: resData }), { headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' } }) } catch (err) { return new Response(JSON.stringify({ error: err.message }), { status: 500, headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' } }) } })
